using UnityEngine; using UnityEngine.UI; public class MobilePlayerController : MonoBehaviour { public float moveSpeed = 4.0f; public Joystick joystick; // Unity UI Joystick Asset public Camera playerCamera; public float touchSensitivity = 0.2f; private float rotationX = 0; private Vector2 touchStartPos; void Update() { // ૧. જોયસ્ટીકથી પ્લેયરનું મુવમેન્ટ (આગળ, પાછળ, ડાબે, જમણે) float moveX = joystick.Horizontal * moveSpeed * Time.deltaTime; float moveZ = joystick.Vertical * moveSpeed * Time.deltaTime; transform.Translate(moveX, 0, moveZ); // ૨. સ્ક્રીન પર આંગળી ફેરવીને કેમેરા ફેરવવો (Swipe to look around) if (Input.touchCount > 0) { Touch touch = Input.GetTouch(0); // જો આંગળી જોયસ્ટીક સિવાયની જગ્યાએ સ્ક્રીન પર ફરે છે if (touch.position.x > Screen.width / 3) { if (touch.phase == TouchPhase.Moved) { float mouseX = touch.deltaPosition.x * touchSensitivity; float mouseY = touch.deltaPosition.y * touchSensitivity; rotationX -= mouseY; rotationX = Mathf.Clamp(rotationX, -90f, 90f); playerCamera.transform.localRotation = Quaternion.Euler(rotationX, 0, 0); transform.Rotate(0, mouseX, 0); } } } } // ૩. સ્ક્રીન પરના 'Interact' બટન દબાવવાથી દરવાજો ખોલવા માટે public void OnInteractButtonPressed() { Ray ray = playerCamera.ViewportPointToRay(new Vector3(0.5f, 0.5f, 0)); RaycastHit hit; if (Physics.Raycast(ray, out hit, 3.0f)) { if (hit.collider.CompareTag("Door")) { hit.collider.GetComponent()?.ToggleDoor(); } } } }
